       ```python
       """Persistent onboarding state (D-16, D-17):
       - save_state / load_state / update_status — helpers вокруг onboarding_sessions table
       - OnboardingCleanupWorker — periodic TTL cleanup (каждые 5 минут)
       
       NB: Telethon TelegramClient объект НЕ сериализуется — он остаётся в in-process
       dict внутри routers/onboarding.py. Здесь только phone_code_hash +
       encrypted_session_string + status для recovery после restart api-контейнера.
       """
       
       import asyncio
       import logging
       import os
       from datetime import datetime, timedelta, timezone
       from typing import Optional
       from uuid import UUID, uuid4
       
       from sqlalchemy import select, text
       from sqlalchemy.ext.asyncio import AsyncSession
       
       from app.database import AsyncSessionLocal
       from app.models import OnboardingSession
       from app.services.encryption import encrypt_session
       
       logger = logging.getLogger(__name__)
       
       ONBOARDING_TTL_MINUTES = 10
       ONBOARDING_CLEANUP_INTERVAL_SEC = int(os.environ.get("ONBOARDING_CLEANUP_INTERVAL", "300"))  # 5 min
       
       
       async def save_state(
           db: AsyncSession,
           workspace_id: UUID,
           phone: str,
           phone_code_hash: str,
           session_string: str,
           role: str = "sender",
           proxy: Optional[dict] = None,
       ) -> UUID:
           """Создаёт onboarding_sessions row, шифрует session_string. Возвращает session_id."""
           if role not in ("sender", "checker"):
               raise ValueError(f"Invalid role: {role}")
           expires_at = datetime.now(timezone.utc) + timedelta(minutes=ONBOARDING_TTL_MINUTES)
           session = OnboardingSession(
               id=uuid4(),
               workspace_id=workspace_id,
               phone=phone,
               phone_code_hash=phone_code_hash,
               encrypted_session_string=encrypt_session(session_string),
               role=role,
               proxy=proxy,
               status="code_sent",
               expires_at=expires_at,
           )
           db.add(session)
           await db.commit()
           await db.refresh(session)
           logger.info(f"📱 onboarding state saved: id={str(session.id)[:8]} phone={phone[:6]}*** role={role}")
           return session.id
       
       
       async def load_state(
           db: AsyncSession,
           session_id: UUID,
           workspace_id: UUID,
       ) -> Optional[OnboardingSession]:
           """Загружает row если workspace матчит и expires_at не прошёл."""
           result = await db.execute(
               select(OnboardingSession).where(
                   OnboardingSession.id == session_id,
                   OnboardingSession.workspace_id == workspace_id,  # TODO(v2-rls)
               )
           )
           row = result.scalars().first()
           if row is None:
               return None
           if row.expires_at < datetime.now(timezone.utc):
               return None
           return row
       
       
       async def update_status(
           db: AsyncSession,
           session_id: UUID,
           status: str,
           encrypted_session_string: Optional[str] = None,
       ) -> None:
           """Обновляет status (+ опционально session_string после Telethon update)."""
           if status not in ("code_sent", "awaiting_2fa", "completed", "failed"):
               raise ValueError(f"Invalid status: {status}")
           params = {"sid": str(session_id), "status": status}
           if encrypted_session_string is not None:
               await db.execute(text("""
                   UPDATE onboarding_sessions
                   SET status = :status, encrypted_session_string = :ess
                   WHERE id = :sid
               """), {**params, "ess": encrypted_session_string})
           else:
               await db.execute(text("""
                   UPDATE onboarding_sessions SET status = :status WHERE id = :sid
               """), params)
           await db.commit()
       
       
       async def delete_session(db: AsyncSession, session_id: UUID) -> None:
           await db.execute(
               text("DELETE FROM onboarding_sessions WHERE id = :sid"),
               {"sid": str(session_id)},
           )
           await db.commit()
       
       
       class OnboardingCleanupWorker:
           """Periodic cleanup of expired onboarding_sessions rows (D-16)."""
           
           def __init__(self):
               self._task: Optional[asyncio.Task] = None
               self._running = False
               self.interval = ONBOARDING_CLEANUP_INTERVAL_SEC
           
           def start(self):
               if self._task is None or self._task.done():
                   self._running = True
                   self._task = asyncio.create_task(self._run(), name="onboarding-cleanup")
                   logger.info(f"OnboardingCleanupWorker started (interval={self.interval}s)")
           
           async def stop(self):
               self._running = False
               if self._task and not self._task.done():
                   self._task.cancel()
                   try:
                       await self._task
                   except asyncio.CancelledError:
                       pass
               logger.info("OnboardingCleanupWorker stopped")
           
           async def _run(self):
               while self._running:
                   try:
                       await self._tick()
                   except asyncio.CancelledError:
                       break
                   except Exception as e:
                       logger.error(f"OnboardingCleanupWorker error: {e}", exc_info=True)
                   await asyncio.sleep(self.interval)
           
           async def _tick(self):
               async with AsyncSessionLocal() as db:
                   result = await db.execute(
                       text("DELETE FROM onboarding_sessions WHERE expires_at < NOW()")
                   )
                   await db.commit()
                   if result.rowcount:
                       logger.info(f"🗑️  OnboardingCleanupWorker: deleted {result.rowcount} expired rows")
       
       
       # Module-level singleton
       onboarding_cleanup_worker = OnboardingCleanupWorker()
       ```

       ```python
       async def _resolve_proxy(db: AsyncSession, workspace_id: UUID, proxy_id: Optional[UUID]) -> Optional[dict]:
           """Загружает proxy из workspace pool. None если proxy_id не задан."""
           if proxy_id is None:
               return None
           result = await db.execute(
               select(ProxyPool).where(
                   ProxyPool.id == proxy_id,
                   ProxyPool.workspace_id == workspace_id,
               )
           )
           proxy_row = result.scalars().first()
           if proxy_row is None:
               raise HTTPException(404, {"code": "PROXY_NOT_FOUND", "message": "Proxy not in workspace pool"})
           return {
               "type": proxy_row.type,
               "host": proxy_row.host,
               "port": proxy_row.port,
               "username": proxy_row.username,
               "password": proxy_row.password,
           }
       
       
       async def _get_or_recover_client(session_row: OnboardingSession) -> TelegramClient:
           """HOT: dict[session_id] OR RECOVERY: decrypt_session + reconnect.
           
           Telethon-client объект не сериализуется (D-17), но `session_string`
           уже содержит DC routing после send_code_request — recover'имый.
           """
           sid_str = str(session_row.id)
           client = _in_process_clients.get(sid_str)
           if client is not None:
               return client
           # Recovery path
           logger.info(f"📱 Recovering Telethon client from DB: session={sid_str[:8]}")
           session_string = decrypt_session(session_row.encrypted_session_string)
           client = make_telegram_client(StringSession(session_string), proxy=session_row.proxy)
           await client.connect()
           _in_process_clients[sid_str] = client
           return client
       
       
       def _drop_in_process(session_id: UUID) -> None:
           _in_process_clients.pop(str(session_id), None)
       
       
       async def _create_sender_from_session(
           db: AsyncSession,
           ctx: AuthCtx,
           session_row: OnboardingSession,
           client: TelegramClient,
           name: Optional[str] = None,
       ) -> Sender:
           """После успешного sign_in — создаём Sender row в workspace юзера."""
           # Получаем session_string и user info из Telethon
           session_string = client.session.save()
           me = await client.get_me()
           slug = f"sender-{me.id}" if me else f"sender-{uuid4().hex[:8]}"
           
           sender = Sender(
               workspace_id=ctx.workspace_id,
               slug=slug,
               name=name or (me.first_name if me else slug),
               phone=session_row.phone,

               role=session_row.role,
               proxy=session_row.proxy,

       def _map_telethon_error(e: Exception) -> HTTPException:
           if isinstance(e, PhoneCodeInvalidError):
               return HTTPException(400, {"code": "PHONE_CODE_INVALID", "message": "Invalid SMS code"})
           if isinstance(e, PhoneCodeExpiredError):
               return HTTPException(400, {"code": "PHONE_CODE_EXPIRED", "message": "SMS code expired"})
           if isinstance(e, PasswordHashInvalidError):
               return HTTPException(400, {"code": "PASSWORD_INVALID", "message": "Invalid 2FA password"})
           if isinstance(e, PhoneNumberBannedError):
               return HTTPException(400, {"code": "PHONE_BANNED", "message": "Phone banned by Telegram"})
           if isinstance(e, FloodWaitError):
               return HTTPException(429, {
                   "code": "FLOOD_WAIT",
                   "message": f"Telegram rate limit: retry after {e.seconds}s",
                   "retry_after": e.seconds,
               })
           return HTTPException(500, {"code": "TELETHON_ERROR", "message": str(e)})

       @router.post("/start", response_model=StartResponse)
       async def start_onboarding(
           request: StartRequest,
           ctx: AuthCtx = Depends(auth_dep),
           db: AsyncSession = Depends(get_db),
       ):
           phone = normalize_to_e164(request.phone)
           if phone is None:
               raise HTTPException(400, {"code": "PHONE_INVALID", "message": "Invalid phone format"})
           role = request.role or "sender"
           if role not in ("sender", "checker"):
               raise HTTPException(400, {"code": "INVALID_ROLE", "message": f"Invalid role: {role}"})
           proxy = await _resolve_proxy(db, ctx.workspace_id, request.proxy_id)
           
           # Init Telethon client (in-process)
           client = make_telegram_client(StringSession(), proxy=proxy)
           try:
               await client.connect()
               sent_code = await client.send_code_request(phone)
           except FloodWaitError as e:
               raise _map_telethon_error(e)
           except Exception as e:
               logger.error(f"start_onboarding failed: {e}", exc_info=True)
               raise HTTPException(500, {"code": "TELETHON_ERROR", "message": str(e)})
           
           # ВАЖНО: сохраняем session_string ПОСЛЕ send_code_request (D-17),
           # уже содержит DC routing — позволяет recovery после restart.
           session_string = client.session.save()
           session_id = await save_state(
               db, ctx.workspace_id, phone, sent_code.phone_code_hash,
               session_string, role=role, proxy=proxy,
           )
           _in_process_clients[str(session_id)] = client
           logger.info(f"📱 Onboarding started: session={str(session_id)[:8]} workspace={ctx.workspace_id}")
           return StartResponse(session_id=session_id, status="code_sent", phone=phone)
       
       
       # ─── POST /onboarding/verify-code ─────────────────────────────────────────
       @router.post("/verify-code")
       async def verify_code(
           request: VerifyCodeRequest,
           ctx: AuthCtx = Depends(auth_dep),
           db: AsyncSession = Depends(get_db),
       ):
           session_row = await load_state(db, request.session_id, ctx.workspace_id)
           if session_row is None:
               raise HTTPException(404, {"code": "SESSION_NOT_FOUND", "message": "Onboarding session not found or expired"})
           if session_row.status not in ("code_sent",):
               raise HTTPException(400, {"code": "INVALID_STATE", "message": f"Session in state {session_row.status}"})
           
           # role в verify-code может переопределить (если юзер передумал)
           if request.role and request.role in ("sender", "checker") and request.role != session_row.role:
               # apply override (mutate row in-memory; persist через update later)
               session_row.role = request.role
               await db.execute(__import__("sqlalchemy").text(
                   "UPDATE onboarding_sessions SET role = :r WHERE id = :sid"
               ), {"r": request.role, "sid": str(session_row.id)})
               await db.commit()
           
           client = await _get_or_recover_client(session_row)
           
           try:
               await client.sign_in(
                   phone=session_row.phone,
                   code=request.code,
                   phone_code_hash=session_row.phone_code_hash,
               )
           except SessionPasswordNeededError:
               # 2FA flow
               new_session = encrypt_session(client.session.save())
               await update_status(db, session_row.id, "awaiting_2fa", encrypted_session_string=new_session)
               logger.info(f"🔐 Onboarding awaiting 2FA: session={str(session_row.id)[:8]}")
               return {"status": "2fa_required", "session_id": str(session_row.id)}
           except (PhoneCodeInvalidError, PhoneCodeExpiredError, PhoneNumberBannedError, FloodWaitError) as e:
               raise _map_telethon_error(e)
           except Exception as e:
               logger.error(f"verify_code unexpected error: {e}", exc_info=True)
               await update_status(db, session_row.id, "failed")
               raise HTTPException(500, {"code": "TELETHON_ERROR", "message": str(e)})
           
           sender = await _create_sender_from_session(db, ctx, session_row, client, name=request.name)
           await update_status(db, session_row.id, "completed")
           _drop_in_process(session_row.id)
           try:
               await client.disconnect()
           except Exception:
               pass
           await delete_session(db, session_row.id)
           return {"status": "success", "sender_id": str(sender.id), "slug": sender.slug}
       ```

       ```python
       async def _reconcile_loop(self):
           """Phase 2 D-18: periodic diff desired senders vs currently connected."""
           logger.info(f"🔄 Reconcile loop started (interval={self.reconcile_interval}s)")
           while self.running:
               try:
                   await asyncio.sleep(self.reconcile_interval)
                   if not self.running:
                       break
                   await self._reconcile_tick()
               except asyncio.CancelledError:
                   break
               except Exception as e:
                   logger.error(f"❌ [reconcile] error: {e}", exc_info=True)
           logger.info("🛑 Reconcile loop stopped")
       
       async def _reconcile_tick(self):
           """Один tick реконсиляции (public для unit-тестов)."""
           desired_list = await self.get_active_senders()
           desired = {str(s["id"]): s for s in desired_list}
           current = set(self._connected_sender_ids)
           
           # NEW: connect новых
           new_ids = set(desired.keys()) - current
           for sid in new_ids:
               s = desired[sid]
               logger.info(f"🔄 [reconcile] connecting sender={s['slug']} workspace={str(s.get('workspace_id',''))[:8]}")
               asyncio.create_task(self.start_client(s))
           
           # REMOVED: disconnect ушедших / paused / errored
           removed_ids = current - set(desired.keys())
           for sid in removed_ids:
               slug = self._slug_by_id(sid)
               logger.info(f"🔄 [reconcile] disconnecting sender_id={sid[:8]} slug={slug} (lifecycle/auth changed)")
               if slug and slug in self.clients:
                   client = self.clients.pop(slug, None)
                   if client and client.is_connected():
                       try:
                           await client.disconnect()
                       except Exception as e:
                           logger.warning(f"🔄 [reconcile] disconnect error for {slug}: {e}")
               self._connected_sender_ids.discard(sid)
               self._proxy_snapshot.pop(sid, None)
           
           # PROXY CHANGE: переподключаем при изменении proxy
           for sid, info in desired.items():
               if sid in current and self._proxy_snapshot.get(sid) != info.get("proxy"):
                   logger.warning(f"🔄 [reconcile] proxy changed for sender={info['slug']}, reconnecting on next tick")
                   slug = info["slug"]
                   client = self.clients.pop(slug, None)
                   if client and client.is_connected():
                       try:
                           await client.disconnect()
                       except Exception:
                           pass
                   self._connected_sender_ids.discard(sid)
                   self._proxy_snapshot.pop(sid, None)
                   # next tick подцепит как new_id
           
           logger.debug(
               f"🔄 [reconcile] tick: connected={len(self._connected_sender_ids)} "
               f"desired={len(desired)} +{len(new_ids)} -{len(removed_ids)}"
           )
       
       def _slug_by_id(self, sender_id: str) -> Optional[str]:
           """Lookup slug по sender_id из self.clients reverse map."""
           # Простой O(N) поиск — обычно <50 sender'ов на инстанс
           for slug, client in list(self.clients.items()):
               # Сравнение через id, который мы сохраняли в start_client
               # NB: можем хранить параллельный dict, но проще через атрибут client.sender_id
               sid = getattr(client, "_sender_id", None)
               if sid == sender_id:
                   return slug
           return None
       ```
