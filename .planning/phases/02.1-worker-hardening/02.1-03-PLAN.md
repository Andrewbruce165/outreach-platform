**Шаг 1 — app/services/contact_check_worker.py: добавить FOR UPDATE OF c SKIP LOCKED.**

В _tick (line 103-130) — изменить SELECT:

```python
async def _tick(self) -> int:
    """Один tick: подобрать pending → resolve → update. Returns processed count.

    Phase 02.1 (CR-08): FOR UPDATE OF c SKIP LOCKED предотвращает race между
    несколькими ContactCheckWorker экземплярами (горизонтальный масштаб).
    Лок держится до commit/rollback текущей транзакции — поскольку мы выходим
    из async with до запроса в checker_service, lock освобождается, но rows
    уже помечены contacts.tg_status='registered'|'not_registered'|'error' в
    _apply_results — следующий tick их не подхватит (фильтр tg_status='pending').
    """
    async with AsyncSessionLocal() as db:
        # Используем transaction context чтобы FOR UPDATE OF держался
        async with db.begin():
            result = await db.execute(
                text(
                    """
                    SELECT c.id AS contact_id,
                           c.workspace_id,
                           c.phone,
                           c.username,
                           s.id AS checker_id,
                           s.slug AS checker_slug,
                           s.session_string,
                           s.proxy
                    FROM contacts c
                    JOIN LATERAL (
                        SELECT id, slug, session_string, proxy
                        FROM senders
                        WHERE workspace_id = c.workspace_id
                          AND role = 'checker'
                          AND auth_status = 'ok'
                        LIMIT 1
                    ) s ON TRUE
                    WHERE c.tg_status = 'pending'
                      AND c.phone IS NOT NULL
                    ORDER BY c.created_at ASC
                    LIMIT :n
                    FOR UPDATE OF c SKIP LOCKED
                    """
                ),
                {"n": self.batch_size},
            )
            rows = result.fetchall()
            # Атомарно помечаем 'processing' — освобождает row lock, но защищает
            # от повторного SELECT в следующем tick'е этого же worker'а.
            if rows:
                contact_ids = [r.contact_id for r in rows]
                await db.execute(
                    text("""
                        UPDATE contacts
                        SET tg_status = 'processing', updated_at = NOW()
                        WHERE id = ANY(:ids)
                    """),
                    {"ids": contact_ids},
                )
            # commit на выходе из async with db.begin()

    if not rows:
        return 0

    # ... остаток метода без изменений ...
```

NB: Нужно добавить 'processing' в CHECK constraint contacts.tg_status. Проверить migration 013:
```
CHECK (tg_status IN ('pending', 'registered', 'not_registered', 'error', 'unchecked'))
```
'processing' нужно либо добавить, либо найти альтернативу. Альтернатива (предпочтительнее, чтобы не менять CHECK): использовать только FOR UPDATE SKIP LOCKED БЕЗ перехода в processing — тогда defence-in-depth достаточно на уровне row-lock. Так как:

- Внутри одной транзакции (`async with db.begin()`) другой worker SKIP LOCKED пропустит rows
- На выходе из транзакции мы УЖЕ вызвали checker и в _apply_results записали финальный статус (registered/not_registered/error)
- Следующий SELECT по `tg_status='pending'` их не подхватит

Поэтому **финальный вариант** — БЕЗ 'processing' статуса, только FOR UPDATE OF c SKIP LOCKED:

ПРОБЛЕМА с этим вариантом: lock освобождается ДО checker_service.check_phones (т.к. транзакция commits). Между этим коммитом и UPDATE в _apply_results другой worker может прийти и подхватить эти же rows. → НУЖНО держать lock дольше: либо до checker, либо использовать 'processing' статус.

**Правильное решение — добавить 'processing' в CHECK constraint через миграцию 014 (или создать 015), либо проще: использовать SELECT FOR UPDATE с лонг-running transaction вокруг checker call:**

```python
async def _tick(self) -> int:
    async with AsyncSessionLocal() as db:
        async with db.begin():
            # SELECT + lock
            result = await db.execute(
                text("""
                    SELECT c.id AS contact_id, c.workspace_id, c.phone, c.username,
                           s.id AS checker_id, s.slug AS checker_slug,
                           s.session_string, s.proxy
                    FROM contacts c
                    JOIN LATERAL (...) s ON TRUE
                    WHERE c.tg_status = 'pending' AND c.phone IS NOT NULL
                    ORDER BY c.created_at ASC
                    LIMIT :n

                """),
                {"n": self.batch_size},

            # worker SKIP LOCKED'нет эти rows и не возьмёт их повторно.
            # NB: НЕ меняем tg_status (CHECK constraint) — используем tg_checked_at
            # как weak claim: SELECT с фильтром
            #   tg_status='pending' AND (tg_checked_at IS NULL OR tg_checked_at < NOW() - INTERVAL '5 minutes')
            # на следующих тиках пропустит recently-claimed rows.
            contact_ids = [r.contact_id for r in rows]
            await db.execute(
                text("""
                    UPDATE contacts
                    SET tg_checked_at = NOW()
                    WHERE id = ANY(:ids)
                """),
                {"ids": contact_ids},
            )
            # commit на выходе из async with db.begin()

    # Дальше — без изменений, кроме _tick фильтра:
    # ... existing code ...
```

И в SELECT добавить guard:
```sql
WHERE c.tg_status = 'pending'
  AND c.phone IS NOT NULL
  AND (c.tg_checked_at IS NULL OR c.tg_checked_at < NOW() - INTERVAL '5 minutes')
```

Это даёт **двойную защиту:**
1. FOR UPDATE OF c SKIP LOCKED (in-transaction lock)
2. tg_checked_at-based claim (in-flight 5-min hold)

```python
import hmac
import time as time_module
from datetime import datetime, timezone

# ─── Phase 02.1 (CR-09): in-process token cache ──────────────────────────────
# В Phase 1 был только bcrypt-per-request (~100ms каждый). Для n8n push на
# /api/v1/contacts это становится CPU bottleneck. Cache: 5-min TTL, in-process
# dict (не shared между containers — это OK, второй container просто прогреет
# cache при первых запросах). На revoke токена — старый cache живёт до 5 мин
# (известное ограничение; для immediate revoke нужен Redis pubsub или DB-poll).
_TOKEN_CACHE: dict[str, tuple["AuthCtx", float]] = {}
_TOKEN_CACHE_TTL_SECONDS = 300.0  # 5 минут
_TOKEN_CACHE_MAX_SIZE = 1024     # bounded LRU-ish (eviction = oldest)


def _cache_get(raw_token: str) -> Optional["AuthCtx"]:
    entry = _TOKEN_CACHE.get(raw_token)
    if entry is None:
        return None
    ctx, expires_at = entry
    if expires_at < time_module.time():
        _TOKEN_CACHE.pop(raw_token, None)
        return None
    return ctx


def _cache_put(raw_token: str, ctx: "AuthCtx") -> None:
    # Простая bounded eviction — если переполнено, дропаем ~10% самых старых
    if len(_TOKEN_CACHE) >= _TOKEN_CACHE_MAX_SIZE:
        sorted_keys = sorted(_TOKEN_CACHE.items(), key=lambda kv: kv[1][1])
        for k, _ in sorted_keys[: max(1, _TOKEN_CACHE_MAX_SIZE // 10)]:
            _TOKEN_CACHE.pop(k, None)
    _TOKEN_CACHE[raw_token] = (ctx, time_module.time() + _TOKEN_CACHE_TTL_SECONDS)
```

```python
async def _verify_api_key(db: AsyncSession, raw_token: str) -> AuthCtx:
    """Verify wsk_<...> token. Phase 02.1 (CR-09) hardening:
    - In-process LRU cache 5-min TTL (избегаем bcrypt-per-request bottleneck для n8n push)
    - hmac.compare_digest для prefix string-equality (constant-time defence-in-depth)
    - last_used_at = datetime.now(timezone.utc) вместо func.now() (clean Python attr)
    """
    if len(raw_token) < 12:
        raise HTTPException(
            status_code=401,
            detail={"code": "API_KEY_INVALID", "message": "Malformed workspace key"},
        )

    # ─── Phase 02.1: cache hit fast path ─────────────────────────────
    cached_ctx = _cache_get(raw_token)
    if cached_ctx is not None:
        return cached_ctx

    prefix = raw_token[:12]

    result = await db.execute(
        select(WorkspaceApiKey).where(
            WorkspaceApiKey.prefix == prefix,
            WorkspaceApiKey.revoked_at.is_(None),
        )
    )
    candidates = result.scalars().all()

    for candidate in candidates:
        # Phase 02.1 (CR-09 issue 1): constant-time prefix string-equality.
        # Сам prefix lookup делал SQL — но повторим compare_digest для
        # защиты-в-глубину против сравнения prefix Python'ом.
        if not hmac.compare_digest(prefix.encode(), candidate.prefix.encode()):
            continue
        match = await asyncio.to_thread(
            bcrypt.checkpw,
            raw_token.encode(),
            candidate.bcrypt_hash.encode(),
        )
        if match:
            # Phase 02.1 (CR-09 issue 3): clean Python datetime, не SQL func.now()
            candidate.last_used_at = datetime.now(timezone.utc)
            await db.commit()

            ctx = AuthCtx(
                workspace_id=candidate.workspace_id,
                user_id=None,
                source="api_key",
                role=None,
            )
            # Cache validated token (5-min TTL)
            _cache_put(raw_token, ctx)

            logger.info(
                f"[auth] api_key matched workspace={candidate.workspace_id} "
                f"prefix={prefix} key_id={str(candidate.id)[:8]}..."
            )
            return ctx

    logger.warning(
        f"[auth] api_key lookup failed prefix={prefix} "
        f"candidates={len(candidates)}"
    )
    raise HTTPException(
        status_code=401,
        detail={
            "code": "API_KEY_INVALID",
            "message": "Invalid or revoked workspace key",
        },
    )
```
