@router.get("/funnel", response_model=FunnelResponse)
async def funnel(
    scope: Literal["workspace", "campaign", "agent", "sender"] = "workspace",
    id: Optional[UUID] = None,
    ctx: AuthCtx = Depends(auth_dep),
    db: AsyncSession = Depends(get_db),
):
    """UI-SPEC §5.3 dashboard Sankey funnel — 5 stage counts (Sent→Replied→Engaged→Lead→Handoff).

    'engaged' stage definition (locked per RESEARCH Pitfall 5):
        COUNT(DISTINCT conversation_id) where >= 2 inbound contact messages AND
        status NOT IN ('lead','handoff','finished','bot_ignored').
    bot_ignored conversations are excluded from every count (Phase 5 Pitfall 8 carry-over).
    """
    if scope != "workspace":
        if id is None:
            raise HTTPException(422, detail={"code": "ID_REQUIRED",
                                              "message": f"id required for scope={scope}"})
        scope_col = _FUNNEL_SCOPE_COLUMNS[scope]
        scope_clause = f"AND {scope_col} = :scope_val"
        params = {"wid": str(ctx.workspace_id), "scope_val": str(id)}
        # Validate cross-workspace 404 — reuse existing helpers (or inline single SELECT).
        _table_for_scope = {"campaign": "campaigns", "agent": "ai_contexts", "sender": "senders"}[scope]
        owns = (await db.execute(text(f"""
            SELECT 1 FROM {_table_for_scope}
            WHERE id = :scope_val AND workspace_id = :wid LIMIT 1
        """), params)).first()
        if owns is None:
            raise HTTPException(404, detail={"code": f"{scope.upper()}_NOT_FOUND",
                                              "message": f"{scope} not in your workspace"})
    else:
        scope_clause = ""
        params = {"wid": str(ctx.workspace_id)}

    # sent: outbound messages joined to conversations (Phase 5 C-01 source-of-truth).
    sent = (await db.execute(text(f"""
        SELECT COUNT(*) FROM messages m
        JOIN conversations c ON c.id = m.conversation_id
        WHERE c.workspace_id = :wid
          AND c.status != 'bot_ignored'
          {scope_clause}
          AND m.direction = 'outbound'
    """), params)).scalar() or 0

    # replied: distinct conversations with >= 1 inbound contact msg.
    replied = (await db.execute(text(f"""
        SELECT COUNT(DISTINCT m.conversation_id) FROM messages m
        JOIN conversations c ON c.id = m.conversation_id
        WHERE c.workspace_id = :wid
          AND c.status != 'bot_ignored'
          {scope_clause}
          AND m.direction = 'inbound' AND m.sent_by = 'contact'
    """), params)).scalar() or 0

    # engaged: >= 2 inbound contact messages AND status NOT IN terminal/bot.
    engaged = (await db.execute(text(f"""
        SELECT COUNT(*) FROM (
            SELECT m.conversation_id
            FROM messages m
            JOIN conversations c ON c.id = m.conversation_id
            WHERE c.workspace_id = :wid
              AND c.status NOT IN ('lead','handoff','finished','bot_ignored')
              {scope_clause}
              AND m.direction = 'inbound' AND m.sent_by = 'contact'
            GROUP BY m.conversation_id
            HAVING COUNT(*) >= 2
        ) e
    """), params)).scalar() or 0

    # lead: conversations with status='lead' (strict EQ per Phase 5 Pitfall 9).
    lead = (await db.execute(text(f"""
        SELECT COUNT(*) FROM conversations c
        WHERE c.workspace_id = :wid
          AND c.status = 'lead'
          {scope_clause}
    """), params)).scalar() or 0

    # handoff: conversations with status='handoff' (or status='manual' — UI-SPEC §8.3 maps both to "Manager").
    # Lock to status='handoff' (the AI-triggered signal) — manual takeover is human-driven, not a funnel stage.
    handoff = (await db.execute(text(f"""
        SELECT COUNT(*) FROM conversations c
        WHERE c.workspace_id = :wid
          AND c.status = 'handoff'
          {scope_clause}
    """), params)).scalar() or 0

    return FunnelResponse(sent=sent, replied=replied, engaged=engaged, lead=lead, handoff=handoff)
```

@router.get("/llm", response_model=LLMAggregatesResponse)
async def llm_aggregates(
    scope: Literal["workspace", "campaign"] = "workspace",
    id: Optional[UUID] = None,
    since: Literal["1d", "7d", "30d", "90d"] = "7d",
    ctx: AuthCtx = Depends(auth_dep),
    db: AsyncSession = Depends(get_db),
):
    """UI-SPEC §5.6 LLM trace tab — total_calls / avg_latency_ms / tokens (in/out) / spend_usd_cents over since-window.

    spend_usd_cents is 0 in v1 (pricing-per-model is v2 work). Token counts are summed across all calls.
    """
    days = _SINCE_DAYS[since]
    if scope == "campaign":
        if id is None:
            raise HTTPException(422, detail={"code": "ID_REQUIRED",
                                              "message": "id required for scope=campaign"})
        owns = (await db.execute(text("""
            SELECT 1 FROM campaigns
            WHERE id = :scope_val AND workspace_id = :wid LIMIT 1
        """), {"wid": str(ctx.workspace_id), "scope_val": str(id)})).first()
        if owns is None:
            raise HTTPException(404, detail={"code": "CAMPAIGN_NOT_FOUND",
                                              "message": "Campaign not in your workspace"})
        scope_clause = "AND lc.campaign_id = :scope_val"
        params = {"wid": str(ctx.workspace_id), "scope_val": str(id), "days": days}
    else:
        scope_clause = ""
        params = {"wid": str(ctx.workspace_id), "days": days}

    row = (await db.execute(text(f"""
        SELECT
            COUNT(*)                                    AS total_calls,
            CAST(ROUND(AVG(lc.latency_ms)) AS INT)      AS avg_latency_ms,
            COALESCE(SUM(lc.prompt_tokens), 0)::INT     AS prompt_tokens,
            COALESCE(SUM(lc.completion_tokens), 0)::INT AS completion_tokens,
            COALESCE(SUM(lc.total_tokens), 0)::INT      AS total_tokens
        FROM llm_calls lc
        WHERE lc.workspace_id = :wid
          AND lc.created_at >= NOW() - (:days || ' days')::INTERVAL
          {scope_clause}
    """), params)).first()

    return LLMAggregatesResponse(
        total_calls=row.total_calls or 0,
        avg_latency_ms=row.avg_latency_ms,
        prompt_tokens=row.prompt_tokens or 0,
        completion_tokens=row.completion_tokens or 0,
        total_tokens=row.total_tokens or 0,
        spend_usd_cents=0,  # v1 stub — RESEARCH §"Backend Gap Map" defers per-model pricing
    )
```

"""Telemetry router (Phase 05.1 — UI-SPEC §9).

Ingest UI-fired events for Core Value KPI tracking. JWT-only (no API-key path —
events are user-context, not integration-context). Storage = local Postgres
table `telemetry_events`. NO PostHog/Segment in v1.

Whitelist enforced at ingest time so unknown events get caught at the edge
(prevents schema drift from typos / agent hallucinations).
"""
import logging
import uuid as _uuid
from datetime import datetime
from typing import Any, Optional
from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

from app.database import get_db
from app.schemas import TelemetryEventIn, CoreValueResponse
from app.utils.auth import AuthCtx, auth_dep

logger = logging.getLogger(__name__)
router = APIRouter(prefix="/api/v1/telemetry", tags=["telemetry"])


# UI-SPEC §9 — exact 15-event whitelist (do not edit without updating UI-SPEC §9 + handoff bundle telemetry-events.md).
_EVENT_WHITELIST = {
    "magic_link_requested",
    "signup_completed",
    "sender_added",
    "contacts_imported",
    "csv_import_completed",
    "agent_created",
    "campaign_created",
    "campaign_launched",

    "workspace_api_key_created",
    "settings_changed",
    "agent_voice_changed",
    "custom_tool_added",
    "dashboard_viewed",
}


@router.post("/events", status_code=202)
async def ingest_event(
    body: TelemetryEventIn,
    ctx: AuthCtx = Depends(auth_dep),
    db: AsyncSession = Depends(get_db),
):
    """Ingest a single telemetry event. Idempotent on event_id (ON CONFLICT DO NOTHING).

    Returns 202 Accepted even on duplicate event_id (matches sendBeacon semantics).
    Whitelist rejection returns 400 — this is a contract error, not a transient one.
    """
    if body.event not in _EVENT_WHITELIST:
        raise HTTPException(
            status_code=400,
            detail={"code": "UNKNOWN_EVENT",
                    "message": f"Event '{body.event}' not in whitelist (UI-SPEC §9)"},
        )

    event_id = body.event_id or _uuid.uuid4()

    # SECURITY: never log the props blob — may contain user PII or tokens.
    logger.info("[telemetry] ingest workspace=%s event=%s event_id=%s",
                ctx.workspace_id, body.event, event_id)

    import json
    await db.execute(text("""
        INSERT INTO telemetry_events
            (event_id, workspace_id, user_id, event, props, client_ts, server_ts)
        VALUES (:eid, :wid, :uid, :event, CAST(:props AS JSONB), :cts, NOW())
        ON CONFLICT (event_id) DO NOTHING
    """), {
        "eid": str(event_id),
        "wid": str(ctx.workspace_id),
        "uid": ctx.user_id,
        "event": body.event,
        "props": json.dumps(body.props or {}, ensure_ascii=False, default=str),
        "cts": body.client_timestamp,
    })
    await db.commit()
    return {"accepted": True, "event_id": str(event_id)}


@router.get("/core-value", response_model=CoreValueResponse)
async def core_value(
    ctx: AuthCtx = Depends(auth_dep),
    db: AsyncSession = Depends(get_db),
):
    """UI-SPEC §9 KPI #9: time_to_first_campaign_seconds = MIN(campaign_launched.server_ts) - MIN(signup_completed.server_ts).

    Returns null fields if the workspace hasn't signed up + launched a campaign yet.
    Target: < 600 seconds (10 minutes) for 80% of new users. Measurement is per-workspace.
    """
    row = (await db.execute(text("""
        WITH t AS (
          SELECT
            (SELECT MIN(server_ts) FROM telemetry_events
              WHERE workspace_id = :wid AND event = 'signup_completed') AS signup_at,
            (SELECT MIN(server_ts) FROM telemetry_events
              WHERE workspace_id = :wid AND event = 'campaign_launched') AS first_launch_at
        )
        SELECT signup_at, first_launch_at,
               CASE WHEN signup_at IS NOT NULL AND first_launch_at IS NOT NULL
                    THEN EXTRACT(EPOCH FROM (first_launch_at - signup_at))::INT
                    ELSE NULL END AS delta_seconds
        FROM t
    """), {"wid": str(ctx.workspace_id)})).first()
    return CoreValueResponse(
        time_to_first_campaign_seconds=row.delta_seconds,
        signup_at=row.signup_at,
        first_launch_at=row.first_launch_at,
    )
