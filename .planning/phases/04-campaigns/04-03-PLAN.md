**Шаг 2. Добавить per-campaign helper:**

В queue.py добавить:
```python
import zoneinfo
from datetime import datetime, timezone as dt_tz


def _campaign_in_working_window(
    *,
    campaign_tz: str,
    work_hour_start: int,
    work_hour_end: int,
    work_days_mask: int,
    now: datetime | None = None,
) -> bool:
    """Per-campaign working hours check (Phase 4 D-08, D-09, D-10).

    Args:
        campaign_tz: IANA timezone (e.g., 'Europe/Moscow').
        work_hour_start: 0-23.
        work_hour_end: 1-24 (exclusive).
        work_days_mask: bitmask Mo=1,Tu=2,We=4,Th=8,Fr=16,Sa=32,Su=64.
        now: datetime to test (default NOW UTC).

    Returns True if now (in campaign's tz) falls in hours window and weekday bit set.
    """
    if now is None:
        now = datetime.now(dt_tz.utc)
    try:
        tz = zoneinfo.ZoneInfo(campaign_tz)
    except (zoneinfo.ZoneInfoNotFoundError, Exception) as e:
        logger.warning(f"Invalid campaign timezone '{campaign_tz}': {e}")
        return False

    local_now = now.astimezone(tz)
    weekday_bit = 1 << local_now.weekday()  # Mo=0 → 1, Tu=1 → 2, ...
    if (work_days_mask & weekday_bit) == 0:
        return False
    if not (work_hour_start <= local_now.hour < work_hour_end):
        return False
    return True
```
