```python
"""Phase 4 D-19 / C-02 / C-03: Mustache-style template rendering for campaign messages.

Supported variables (case-insensitive):
- {{name}} / {{имя}}      → contact.full_name
- {{username}} / {{юзернейм}} → '@' + contact.username (empty if no username)
- {{phone}} / {{телефон}} → contact.phone
- {{source}} / {{источник}} → contact.source
- {{custom.X}}            → contact.custom[X]  (X is English-only)
- {{компания}}            → contact.custom['company']  (alias to custom.company)

Spaces inside braces OK: {{ name }} works.
Missing variables → empty string + logger.warning (D-19; strict mode deferred to v2).
No Mustache filters ({{name | upper}}) per C-03.
"""

import logging
import re
from typing import Any, Optional

logger = logging.getLogger(__name__)

# Regex per C-03: allows spaces inside braces, supports Cyrillic letters,
# dot notation for custom.X (English keys only).
TEMPLATE_VAR_RE = re.compile(
    r"\{\{\s*([a-zA-Zа-яА-Я_][a-zA-Zа-яА-Я_0-9]*(?:\.[a-zA-Z_0-9]+)?)\s*\}\}",
    re.IGNORECASE | re.UNICODE,
)

# C-02 — Russian alias table (lowercase keys → canonical English var name)
RUSSIAN_ALIASES = {
    "имя": "name",
    "юзернейм": "username",
    "телефон": "phone",
    "источник": "source",
    "компания": "custom.company",
}


def _resolve(var_name: str, contact: dict[str, Any]) -> Optional[str]:
    """Resolve {{var}} → contact field value. Returns None if missing/unknown.

    Args:
        var_name: variable name from inside braces (already lowercased).
        contact: dict with contact fields (full_name, username, phone, source, custom).

    Returns: string value or None when variable cannot be resolved.
    """
    # Apply Russian alias
    if var_name in RUSSIAN_ALIASES:
        var_name = RUSSIAN_ALIASES[var_name]

    # custom.X dotted nodes
    if "." in var_name:
        prefix, _, key = var_name.partition(".")
        if prefix != "custom":
            return None
        custom = contact.get("custom") or {}
        value = custom.get(key)
        return None if value is None else str(value)

    if var_name == "name":
        v = contact.get("full_name")
        return None if not v else str(v)
    if var_name == "username":
        v = contact.get("username")
        return None if not v else f"@{v}"
    if var_name == "phone":
        v = contact.get("phone")
        return None if not v else str(v)
    if var_name == "source":
        v = contact.get("source")
        return None if not v else str(v)

    return None


def render_template(
    template: str,
    contact: dict[str, Any],
    *,
    campaign_id: str = "?",
    phone: str = "?",
) -> str:
    """Render Mustache-style template with contact fields.

    Args:
        template: string with {{var}} placeholders.
        contact: dict with full_name / username / phone / source / custom.
        campaign_id: for warning logs (no semantic effect on render).
        phone: for warning logs (no semantic effect on render).

    Returns: rendered string (missing vars → '' + warning).
    """
    if not template:
        return ""

    def replacer(match: re.Match) -> str:
        raw = match.group(1)
        # case-insensitive lookup — lower before resolve
        value = _resolve(raw.lower(), contact)
        if value is None:
            logger.warning(
                "Template variable {{%s}} missing for contact phone=%s campaign=%s",
                raw, phone, campaign_id,
            )
            return ""
        return value

    return TEMPLATE_VAR_RE.sub(replacer, template)

async def get_or_assign_sender(
    campaign_id: UUID,
    contact_phone: str,
    db: AsyncSession,
) -> Optional[Sender]:
    """Phase 4 D-06: rotation per-campaign.

    1. Lookup existing assignment in campaign_contact_assignments — if found, return sender (idempotent).
    2. Otherwise SELECT active senders from campaign_senders pool (auth_status='ok' AND lifecycle_status='active').
    3. Pick least-loaded sender (existing _pick_best_sender pattern adapted).
    4. INSERT into campaign_contact_assignments with workspace_id from campaign.
       ON CONFLICT (campaign_id, contact_phone) DO NOTHING — race-safe.
    5. Return picked sender; or None if no active senders в кампании (caller обработает).

    Args:
        campaign_id: target campaign.
        contact_phone: E.164 phone of contact.
        db: AsyncSession.

    Returns: Sender or None.

    # TODO(v2-rls): replaced by RLS policy app.workspace_id
    """
    # 1. Lookup workspace_id and existing assignment via single query
    existing = await db.execute(text("""
        SELECT cca.sender_id, c.workspace_id
        FROM campaign_contact_assignments cca
        JOIN campaigns c ON c.id = cca.campaign_id
        WHERE cca.campaign_id = :cid AND cca.contact_phone = :phone
    """), {"cid": str(campaign_id), "phone": contact_phone})
    row = existing.fetchone()
    if row:
        sender = (await db.execute(select(Sender).where(Sender.id == row.sender_id))).scalars().first()
        if sender and sender.auth_status == "ok" and sender.lifecycle_status == "active":
            return sender
        # else: existing sender dropped offline — реассайнить ниже

    # 2. Get campaign workspace_id
    camp_row = (await db.execute(text("SELECT workspace_id FROM campaigns WHERE id=:cid"), {"cid": str(campaign_id)})).fetchone()
    if not camp_row:
        return None
    workspace_id = camp_row.workspace_id

    # 3. SELECT active senders from campaign_senders pool
    # Phase 02.1 CR-03: explicit workspace_id guard
    candidates_rows = await db.execute(text("""
        SELECT s.id, s.workspace_id
        FROM campaign_senders cs
        JOIN senders s ON s.id = cs.sender_id
        WHERE cs.campaign_id = :cid
          AND s.auth_status = 'ok'
          AND s.lifecycle_status = 'active'
          AND s.workspace_id = :wid  -- defence-in-depth
    """), {"cid": str(campaign_id), "wid": str(workspace_id)})
    candidates = [r.id for r in candidates_rows.fetchall()]
    if not candidates:
        logger.warning(f"No active senders in campaign {campaign_id}")
        return None

    # 4. Pick least-loaded sender (existing helper or simple impl).
    # Reuse существующий _pick_best_sender если он принимает list[UUID]; иначе round-robin по counts:
    picked_id = await _pick_least_loaded(db, candidates)  # implement или adapt
    sender = (await db.execute(select(Sender).where(Sender.id == picked_id))).scalars().first()

    # 5. INSERT cca with ON CONFLICT DO NOTHING (race-safe)
    if row and not (sender.auth_status == "ok" and sender.lifecycle_status == "active"):
        # Existing assignment is stale (sender died) — UPDATE
        await db.execute(text("""
            UPDATE campaign_contact_assignments
            SET sender_id = :sid
            WHERE campaign_id = :cid AND contact_phone = :phone
        """), {"sid": str(picked_id), "cid": str(campaign_id), "phone": contact_phone})
    else:
        await db.execute(text("""
            INSERT INTO campaign_contact_assignments (workspace_id, campaign_id, contact_phone, sender_id)
            VALUES (:wid, :cid, :phone, :sid)
            ON CONFLICT (campaign_id, contact_phone) DO NOTHING
        """), {"wid": str(workspace_id), "cid": str(campaign_id), "phone": contact_phone, "sid": str(picked_id)})

    await db.commit()
    return sender


async def _pick_least_loaded(db: AsyncSession, sender_ids: list[UUID]) -> UUID:
    """Pick sender_id с минимальным числом active assignments (load-balancing)."""
    rows = await db.execute(text("""
        SELECT s.id AS sid, COUNT(cca.id) AS cnt
        FROM senders s
        LEFT JOIN campaign_contact_assignments cca ON cca.sender_id = s.id
        WHERE s.id = ANY(:ids)
        GROUP BY s.id
        ORDER BY cnt ASC
        LIMIT 1
    """), {"ids": [str(i) for i in sender_ids]})
    r = rows.fetchone()
    return r.sid if r else sender_ids[0]
```

"""Phase 4 D-17: CampaignEnqueueWorker — background generator queue items per running campaign.

Pattern: ContactCheckWorker (Phase 2).
Singleton instance, start/stop in FastAPI lifespan.
"""

import asyncio
import logging
from datetime import datetime, timezone
from typing import Optional
from uuid import UUID

from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

from app.config import settings
from app.database import AsyncSessionLocal
from app.services.rotation import get_or_assign_sender
from app.services.template import render_template

logger = logging.getLogger(__name__)


class CampaignEnqueueWorker:
    """Background worker generating queue items from running campaigns' folder contacts."""

    def __init__(self):
        self._task: Optional[asyncio.Task] = None
        self._running = False
        self.batch_size = settings.campaign_enqueue_batch_size
        self.poll_interval = settings.campaign_enqueue_tick_seconds

    def start(self):
        if self._task is None or self._task.done():
            self._running = True
            self._task = asyncio.create_task(self._run(), name="campaign-enqueue-worker")
            logger.info(
                "📤 CampaignEnqueueWorker started (batch=%s, poll=%ss)",
                self.batch_size, self.poll_interval,
            )

    async def stop(self):
        self._running = False
        if self._task and not self._task.done():
            self._task.cancel()
            try:
                await self._task
            except asyncio.CancelledError:
                pass
        logger.info("📤 CampaignEnqueueWorker stopped")

    async def _run(self):
        while self._running:
            try:
                await self._tick()
            except asyncio.CancelledError:
                break
            except Exception as exc:
                logger.error("❌ CampaignEnqueueWorker tick error: %s", exc, exc_info=True)
            await asyncio.sleep(self.poll_interval)

    async def _tick(self) -> int:
        """Returns count of enqueued items."""
        async with AsyncSessionLocal() as db:
            # 1. SELECT running campaigns (no SKIP LOCKED — single-process API container per v1)
            campaigns_rows = await db.execute(text("""
                SELECT id, workspace_id, folder_id, message_template, start_date
                FROM campaigns
                WHERE status = 'running'
            """))
            campaigns = campaigns_rows.fetchall()
            total_enqueued = 0
            for c in campaigns:
                enqueued = await self._tick_one_campaign(db, c)
                total_enqueued += enqueued
            if total_enqueued > 0:
                logger.info("📤 CampaignEnqueueWorker tick: enqueued %s items across %s campaigns", total_enqueued, len(campaigns))
            return total_enqueued

    async def _tick_one_campaign(self, db: AsyncSession, c) -> int:
        """Process one campaign per tick. Atomic per-batch transaction.

        Phase 02.1 CR-03 pattern: explicit workspace_id guard in JOINs.
        Q5: INSERT cca + INSERT message_queue in one transaction — rollback if any fails.
        """
        # 2. SELECT eligible contacts (Pitfall 8 workspace_id guard в WHERE)
        contacts_rows = await db.execute(text("""
            SELECT id, phone, full_name, username, source, custom, workspace_id, folder_id
            FROM contacts
            WHERE folder_id = :fid
              AND workspace_id = :wid
              AND tg_status = 'registered'
              AND phone NOT IN (
                  SELECT contact_phone FROM campaign_contact_assignments
                  WHERE campaign_id = :cid
              )
            LIMIT :limit
        """), {
            "fid": str(c.folder_id),
            "wid": str(c.workspace_id),
            "cid": str(c.id),
            "limit": self.batch_size,
        })
        contacts = contacts_rows.fetchall()
        if not contacts:
            return 0

        now_utc = datetime.now(timezone.utc)
        scheduled_at = max(now_utc, c.start_date) if c.start_date else now_utc

        enqueued = 0
        for contact in contacts:
            try:
                # Q5: atomic per-contact transaction (savepoint within outer)
                async with db.begin_nested():
                    # 3. Rotation — assign sender (commits cca internally via ON CONFLICT)
                    sender = await get_or_assign_sender(c.id, contact.phone, db)
                    if sender is None:
                        logger.warning("CampaignEnqueueWorker: no sender for contact %s in campaign %s", contact.phone, c.id)
                        continue

                    # 4. Render template
                    contact_dict = {
                        "full_name": contact.full_name,
                        "username": contact.username,
                        "phone": contact.phone,
                        "source": contact.source,
                        "custom": contact.custom or {},
                    }
                    rendered = render_template(
                        c.message_template,
                        contact_dict,
                        campaign_id=str(c.id),
                        phone=contact.phone,
                    )

                    # 5. INSERT queue item
                    await db.execute(text("""
                        INSERT INTO message_queue
                          (workspace_id, campaign_id, sender_id, recipient_phone, recipient_name,
                           message_text, scheduled_at, status, created_at)
                        VALUES (:wid, :cid, :sid, :phone, :name, :text, :scheduled, 'pending', NOW())
                    """), {
                        "wid": str(c.workspace_id),
                        "cid": str(c.id),
                        "sid": str(sender.id),
                        "phone": contact.phone,
                        "name": contact.full_name or "",
                        "text": rendered,
                        "scheduled": scheduled_at,
                    })
                    enqueued += 1
            except Exception as e:
                # savepoint rolled back — try next contact
                logger.error("CampaignEnqueueWorker: error enqueuing contact %s in campaign %s: %s", contact.phone, c.id, e, exc_info=True)
                continue

        await db.commit()
        return enqueued


campaign_enqueue_worker = CampaignEnqueueWorker()

**Шаг 2. Рерайт `app/routers/send.py`:**

```python
import logging
from datetime import datetime, timezone
from typing import Optional
from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException
from sqlalchemy import select, text
from sqlalchemy.ext.asyncio import AsyncSession

from app.database import get_db
from app.models import Campaign, Sender
from app.schemas import SendMessageRequest, SendMessageResponse  # SendMessageResponse — добавить если нет
from app.services.rotation import get_or_assign_sender
from app.services.template import render_template
from app.utils.auth import AuthCtx, auth_dep

logger = logging.getLogger(__name__)

router = APIRouter(prefix="/api/v1/send", tags=["send"])


@router.post("", response_model=SendMessageResponse, status_code=202)
async def send_message(
    payload: SendMessageRequest,
    ctx: AuthCtx = Depends(auth_dep),
    db: AsyncSession = Depends(get_db),
):
    """Phase 4 D-16: campaign-based send.

    1. Workspace-scoped campaign fetch (404 if not in ctx.workspace_id).
    2. If sender_id provided — validate workspace; else rotation.get_or_assign_sender.
    3. If text not provided — render_template(campaign.message_template, contact lookup by phone).
    4. INSERT message_queue with campaign_id.
    """
    # 1. Workspace-scoped campaign lookup
    campaign = (await db.execute(
        select(Campaign).where(
            Campaign.id == payload.campaign_id,
            Campaign.workspace_id == ctx.workspace_id,
        )
    )).scalars().first()
    if campaign is None:
        raise HTTPException(404, {"code": "CAMPAIGN_NOT_FOUND", "message": "Campaign not in your workspace"})

    # 2. Sender resolve
    if payload.sender_id:
        sender = (await db.execute(
            select(Sender).where(
                Sender.id == payload.sender_id,
                Sender.workspace_id == ctx.workspace_id,
            )
        )).scalars().first()
        if sender is None:
            raise HTTPException(404, {"code": "SENDER_NOT_FOUND"})
    else:
        sender = await get_or_assign_sender(campaign.id, payload.recipient_phone, db)
        if sender is None:
            raise HTTPException(409, {"code": "NO_ACTIVE_SENDER_IN_CAMPAIGN"})

    # 3. Render text if not provided
    if payload.text:
        rendered_text = payload.text
    else:
        # Lookup contact by phone (workspace-scoped)
        contact_row = (await db.execute(text("""
            SELECT phone, full_name, username, source, custom
            FROM contacts
            WHERE phone = :phone AND workspace_id = :wid
            LIMIT 1
        """), {"phone": payload.recipient_phone, "wid": str(ctx.workspace_id)})).fetchone()
        contact_dict = {
            "phone": contact_row.phone if contact_row else payload.recipient_phone,
            "full_name": (contact_row.full_name if contact_row else payload.recipient_name) or "",
            "username": contact_row.username if contact_row else None,
            "source": contact_row.source if contact_row else None,
            "custom": (contact_row.custom or {}) if contact_row else {},
        }
        rendered_text = render_template(campaign.message_template, contact_dict, campaign_id=str(campaign.id), phone=payload.recipient_phone)

    # 4. INSERT queue
    scheduled = payload.scheduled_at or datetime.now(timezone.utc)
    queue_row = await db.execute(text("""
        INSERT INTO message_queue
            (workspace_id, campaign_id, sender_id, recipient_phone, recipient_name, message_text, scheduled_at, status, created_at)
        VALUES (:wid, :cid, :sid, :phone, :name, :text, :sched, 'pending', NOW())
        RETURNING id
    """), {
        "wid": str(ctx.workspace_id), "cid": str(campaign.id), "sid": str(sender.id),
        "phone": payload.recipient_phone, "name": payload.recipient_name or "",
        "text": rendered_text, "sched": scheduled,
    })
    queue_id = queue_row.scalar()
    await db.commit()

    return SendMessageResponse(queue_id=queue_id, scheduled_at=scheduled)
```
