```python
"""Phase 4 test util — mock OpenAI Chat Completions response with tool_calls."""

from dataclasses import dataclass, field
from typing import Any
from unittest.mock import AsyncMock


@dataclass
class MockToolCall:
    id: str
    function_name: str
    function_arguments: str  # JSON string

    @property
    def function(self):
        class F:
            name = self.function_name
            arguments = self.function_arguments
        return F()


@dataclass
class MockResponseMessage:
    content: str | None
    tool_calls: list | None = None


@dataclass
class MockChatChoice:
    message: MockResponseMessage


@dataclass
class MockChatResponse:
    choices: list[MockChatChoice]


def make_openai_response(
    *,
    text_content: str | None = None,
    tool_calls: list[dict] | None = None,
) -> MockChatResponse:
    """Build a mock OpenAI Chat Completions response.

    Args:
        text_content: response_message.content (для Q3 text + tool_call случая).
        tool_calls: list of {"name": "...", "arguments": "{...}"} dicts.

    Returns: MockChatResponse with single choice.
    """
    mocked_tool_calls = None
    if tool_calls:
        mocked_tool_calls = [
            MockToolCall(id=f"call_{i}", function_name=tc["name"], function_arguments=tc["arguments"])
            for i, tc in enumerate(tool_calls)
        ]
    return MockChatResponse(
        choices=[MockChatChoice(message=MockResponseMessage(content=text_content, tool_calls=mocked_tool_calls))]
    )


def patched_openai_client(monkeypatch, return_response: MockChatResponse):
    """Helper to monkey-patch ai_engine's OpenAI client.chat.completions.create."""
    async def fake_create(*args, **kwargs):
        return return_response

    monkeypatch.setattr(
        "app.services.ai_engine.client.chat.completions.create",
        AsyncMock(side_effect=lambda *a, **k: fake_create(*a, **k)),
    )
```

```python
"""Phase 4 D-13: Fire-and-forget webhook helper for built-in signals (lead/handoff/finish).

Uniform payload shape per C-01. No HMAC (deferred to v2).
Pattern: queue.py:_fire_callback async fire-and-forget.
"""

import asyncio
import logging
from datetime import datetime, timezone
from typing import Any
from uuid import UUID

import httpx
from sqlalchemy import text
from sqlalchemy.ext.asyncio import AsyncSession

logger = logging.getLogger(__name__)

WEBHOOK_TIMEOUT_SECONDS = 30
MESSAGE_HISTORY_LIMIT = 20  # C-01 — last 20 messages в excerpt


async def _fetch_history_excerpt(db: AsyncSession, conversation_id: UUID) -> list[dict]:
    """SELECT last 20 messages of conversation (chronologically asc)."""
    rows = await db.execute(text("""
        SELECT direction, message_text, created_at
        FROM messages_log
        WHERE conversation_id = :cid
        ORDER BY created_at DESC
        LIMIT :limit
    """), {"cid": str(conversation_id), "limit": MESSAGE_HISTORY_LIMIT})
    messages = []
    for r in rows.fetchall():
        role = "assistant" if r.direction == "outbound" else "user"
        messages.append({
            "role": role,
            "content": r.message_text or "",
            "timestamp": r.created_at.isoformat() if r.created_at else None,
        })
    return list(reversed(messages))  # chronological asc for downstream


async def _fire(url: str, payload: dict) -> None:
    """Fire-and-forget POST. Catches all exceptions."""
    try:
        async with httpx.AsyncClient(timeout=WEBHOOK_TIMEOUT_SECONDS) as client:
            r = await client.post(url, json=payload)
            logger.info("Webhook fired event=%s url=%s status=%s", payload.get("event_type"), url, r.status_code)
    except Exception as e:
        logger.warning("Webhook failed event=%s url=%s err=%s", payload.get("event_type"), url, e)


async def notify_signal(
    *,
    event_type: str,
    campaign: dict[str, Any],
    conversation_id: UUID,
    contact: dict[str, Any],
    reason: str,
    db: AsyncSession,
) -> None:
    """Build payload and fire webhook (asyncio.create_task — non-blocking).

    Args:
        event_type: 'lead' | 'handoff' | 'finish'.
        campaign: dict with id, name, workspace_id, {event_type}_webhook_url.
        conversation_id: UUID.
        contact: dict with phone, telegram_id, name (=full_name), username, source, custom.
        reason: LLM-supplied reason string.
        db: AsyncSession (for message history excerpt).
    """
    if event_type not in ("lead", "handoff", "finish"):
        logger.error("notify_signal: invalid event_type=%s", event_type)
        return

    url_key = f"{event_type}_webhook_url"
    url = campaign.get(url_key)
    if not url:
        logger.info("notify_signal: %s URL is NULL for campaign %s — skip", url_key, campaign.get("id"))
        return

    excerpt = await _fetch_history_excerpt(db, conversation_id)

    payload = {
        "event_type": event_type,
        "campaign_id": str(campaign["id"]),
        "campaign_name": campaign.get("name"),
        "conversation_id": str(conversation_id),
        "workspace_id": str(campaign.get("workspace_id")),
        "contact": {
            "phone": contact.get("phone"),
            "telegram_id": contact.get("telegram_id"),
            "name": contact.get("full_name") or contact.get("name"),
            "username": contact.get("username"),
            "source": contact.get("source"),
            "custom": contact.get("custom") or {},
        },
        "reason": reason or "",
        "message_history_excerpt": excerpt,
        "timestamp": datetime.now(timezone.utc).isoformat(),
    }

    # Fire-and-forget (Pattern Pitfall: NEVER inside DB transaction)
    asyncio.create_task(_fire(url, payload))
```

```python
async def get_context_for_conversation(conversation_id: UUID, db: AsyncSession) -> dict | None:
    """Phase 4: resolve conversation → campaign → agent context.

    Returns merged dict с агентскими полями (system_prompt, rules, etc) + campaign-level
    tools / *_trigger_hint / *_webhook_url / message_template / name / workspace_id.

    Если conversation.campaign_id IS NULL (legacy Phase 3 conversations) — возвращает только
    agent context без tools/hints/webhooks.
    """
    row = (await db.execute(text("""
        SELECT
            conv.id AS conv_id, conv.workspace_id AS conv_wid, conv.campaign_id, conv.ai_context_id,
            a.id AS agent_id, a.name AS agent_name, a.system_prompt, a.rules, a.tone_of_voice,
            a.faq, a.company_info, a.product_info,
            c.id AS campaign_id_full, c.name AS campaign_name, c.workspace_id AS campaign_wid,
            c.tools AS campaign_tools, c.message_template,
            c.lead_trigger_hint, c.handoff_trigger_hint, c.finish_trigger_hint,
            c.lead_webhook_url, c.handoff_webhook_url, c.finish_webhook_url
        FROM conversations conv
        LEFT JOIN campaigns c ON c.id = conv.campaign_id
        LEFT JOIN ai_contexts a ON a.id = COALESCE(c.agent_id, conv.ai_context_id)
        WHERE conv.id = :cid
    """), {"cid": str(conversation_id)})).fetchone()

    if not row:
        return None

    context = {
        "conversation_id": row.conv_id,
        "workspace_id": row.conv_wid,
        # Agent fields:
        "agent_id": row.agent_id,
        "agent_name": row.agent_name,
        "system_prompt": row.system_prompt or "",
        "rules": row.rules or "",
        "tone_of_voice": row.tone_of_voice or "",
        "faq": row.faq or {},
        "company_info": row.company_info or "",
        "product_info": row.product_info or "",
    }

    if row.campaign_id_full is not None:
        context["campaign"] = {
            "id": row.campaign_id_full,
            "name": row.campaign_name,
            "workspace_id": row.campaign_wid,
            "tools": row.campaign_tools or [],
            "message_template": row.message_template,
            "lead_trigger_hint": row.lead_trigger_hint,
            "handoff_trigger_hint": row.handoff_trigger_hint,
            "finish_trigger_hint": row.finish_trigger_hint,
            "lead_webhook_url": row.lead_webhook_url,
            "handoff_webhook_url": row.handoff_webhook_url,
            "finish_webhook_url": row.finish_webhook_url,
        }
    else:
        # Legacy conversation без campaign_id — empty campaign context
        context["campaign"] = None

    return context


async def _handle_builtin_signal(
    *,
    db: AsyncSession,
    conversation_id: UUID,
    campaign: dict,
    contact: dict,
    signal_name: str,
    reason: str,
) -> str:
    """Dispatch built-in signal: UPDATE conversation.status + fire webhook.

    Returns final status ('lead' | 'handoff' | 'finished').
    """
    now = datetime.now(timezone.utc)
    if signal_name == "mark_as_lead":
        new_status = "lead"
        await db.execute(text("""
            UPDATE conversations SET status='lead', updated_at=NOW()
            WHERE id = :cid
        """), {"cid": str(conversation_id)})
        await db.commit()
        await notify_signal(event_type="lead", campaign=campaign, conversation_id=conversation_id, contact=contact, reason=reason, db=db)
        return "lead"

    elif signal_name == "transfer_to_manager":
        new_status = "handoff"
        await db.execute(text("""
            UPDATE conversations SET status='handoff', ai_enabled=false, paused_at=:now, paused_reason=:reason, updated_at=NOW()
            WHERE id = :cid
        """), {"cid": str(conversation_id), "now": now, "reason": reason[:200]})
        await db.commit()
        await notify_signal(event_type="handoff", campaign=campaign, conversation_id=conversation_id, contact=contact, reason=reason, db=db)
        return "handoff"

    elif signal_name == "finish_conversation":
        new_status = "finished"
        await db.execute(text("""
            UPDATE conversations SET status='finished', ai_enabled=false, paused_at=:now, paused_reason=:reason, updated_at=NOW()
            WHERE id = :cid
        """), {"cid": str(conversation_id), "now": now, "reason": reason[:200]})
        await db.commit()
        await notify_signal(event_type="finish", campaign=campaign, conversation_id=conversation_id, contact=contact, reason=reason, db=db)
        return "finished"

    logger.error("_handle_builtin_signal: unknown signal_name=%s", signal_name)
    return ""
```
